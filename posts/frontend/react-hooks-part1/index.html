React Hooks 모음 1

📅2025년 11월 24일
프론트엔드ReactHooksJavaScript

React Hooks는 함수형 컴포넌트에서 상태 관리와 사이드 이펙트를 다룰 수 있게 해주는 강력한 도구입니다.

이 글에서는 가장 자주 사용하는 4가지 기본 Hooks를 실전 예제와 함께 살펴봅니다.

useState - 상태 관리의 기본

가장 기본적이면서도 가장 많이 사용하는 Hook입니다.

기본 사용법

import { useState } from 'react';

function Counter() {
  const [count, setCount] = useState(0);

  return (
    <div>
      <p>Count: {count}</p>
      <button onClick={() => setCount(count + 1)}>+</button>
      <button onClick={() => setCount(count - 1)}>-</button>
    </div>
  );
}

핵심 개념:

  • useState는 [현재값, 업데이트함수] 배열을 반환
  • 초기값은 첫 렌더링 때만 사용됨
  • 상태가 변경되면 컴포넌트가 리렌더링됨

함수형 업데이트

상태 업데이트를 연속으로 두 번 호출하면 어떻게 될까요? 값이 아닌 함수를 전달하는 '함수형 업데이트'를 사용하면 이런 문제를 해결할 수 있습니다.

function Counter() {
  const [count, setCount] = useState(0);

  // 나쁜 예: 이전 상태에 의존하는 경우
  const handleIncrement = () => {
    setCount(count + 1);
    setCount(count + 1); // 두 번 증가? 아니요, 한 번만 증가!
  };

  // 좋은 예: 함수형 업데이트
  const handleIncrementTwice = () => {
    setCount(prev => prev + 1);
    setCount(prev => prev + 1); // 제대로 두 번 증가
  };

  return (
    <div>
      <p>Count: {count}</p>
      <button onClick={handleIncrement}>+1 (잘못된 방법)</button>
      <button onClick={handleIncrementTwice}>+2 (올바른 방법)</button>
    </div>
  );
}

왜 함수형 업데이트를 써야 할까?

  • 이전 상태를 기반으로 업데이트할 때 안전
  • useEffect 의존성 배열에서 상태를 제외 가능
  • 클로저 문제 해결

객체 상태 관리

객체 타입의 상태를 업데이트할 때는 주의가 필요합니다. React는 얕은 비교(shallow comparison)를 하기 때문에 객체의 일부만 변경하면 나머지 필드가 사라집니다.

function UserForm() {
  const [user, setUser] = useState({
    name: '',
    email: '',
    age: 0
  });

  // 나쁜 예: 일부만 업데이트하면 나머지 사라짐
  const handleNameChange = (e) => {
    setUser({ name: e.target.value }); // email과 age가 사라짐!
  };

  // 좋은 예: spread 연산자로 병합
  const handleChange = (field) => (e) => {
    setUser(prev => ({
      ...prev,
      [field]: e.target.value
    }));
  };

  return (
    <form>
      <input
        value={user.name}
        onChange={handleChange('name')}
        placeholder="Name"
      />
      <input
        value={user.email}
        onChange={handleChange('email')}
        placeholder="Email"
      />
      <input
        type="number"
        value={user.age}
        onChange={handleChange('age')}
        placeholder="Age"
      />
      <p>{JSON.stringify(user, null, 2)}</p>
    </form>
  );
}

배열 상태 관리

배열 상태를 다룰 때는 원본을 직접 수정하지 말고, 새로운 배열을 만들어 반환해야 합니다. 실무에서 가장 많이 사용하는 CRUD 패턴을 살펴보겠습니다.

function TodoList() {
  const [todos, setTodos] = useState([]);
  const [input, setInput] = useState('');

  // 추가
  const addTodo = () => {
    if (!input.trim()) return;
    setTodos(prev => [...prev, {
      id: Date.now(),
      text: input,
      completed: false
    }]);
    setInput('');
  };

  // 토글
  const toggleTodo = (id) => {
    setTodos(prev => prev.map(todo =>
      todo.id === id
        ? { ...todo, completed: !todo.completed }
        : todo
    ));
  };

  // 삭제
  const deleteTodo = (id) => {
    setTodos(prev => prev.filter(todo => todo.id !== id));
  };

  // 수정
  const updateTodo = (id, newText) => {
    setTodos(prev => prev.map(todo =>
      todo.id === id
        ? { ...todo, text: newText }
        : todo
    ));
  };

  return (
    <div>
      <input
        value={input}
        onChange={(e) => setInput(e.target.value)}
        onKeyPress={(e) => e.key === 'Enter' && addTodo()}
      />
      <button onClick={addTodo}>Add</button>
      <ul>
        {todos.map(todo => (
          <li key={todo.id}>
            <input
              type="checkbox"
              checked={todo.completed}
              onChange={() => toggleTodo(todo.id)}
            />
            <span
              style={{
                textDecoration: todo.completed ? 'line-through' : 'none'
              }}
            >
              {todo.text}
            </span>
            <button onClick={() => deleteTodo(todo.id)}>Delete</button>
          </li>
        ))}
      </ul>
    </div>
  );
}

Lazy Initialization

초기값 계산이 비용이 큰 경우, 함수를 전달하면 첫 렌더링 때만 실행됩니다.

function ExpensiveComponent() {
  // 매번 실행됨
  const [data, setData] = useState(computeExpensiveValue());

  // 첫 렌더링 때만 실행
  const [data, setData] = useState(() => computeExpensiveValue());

  // 실무 예시: localStorage에서 읽기
  const [user, setUser] = useState(() => {
    const saved = localStorage.getItem('user');
    return saved ? JSON.parse(saved) : null;
  });

  return <div>{/* ... */}</div>;
}

useEffect - 사이드 이펙트 처리

컴포넌트가 렌더링된 후 실행되는 코드를 작성합니다.

기본 사용법

import { useState, useEffect } from 'react';

function PageTitle() {
  const [count, setCount] = useState(0);

  // 매 렌더링마다 실행
  useEffect(() => {
    document.title = `Count: ${count}`;
  });

  // 마운트 시에만 실행 (빈 배열)
  useEffect(() => {
    console.log('Component mounted');
  }, []);

  // count가 변경될 때만 실행
  useEffect(() => {
    document.title = `Count: ${count}`;
  }, [count]);

  return (
    <div>
      <p>Count: {count}</p>
      <button onClick={() => setCount(c => c + 1)}>Increment</button>
    </div>
  );
}

의존성 배열 규칙:

  • 없음: 매 렌더링마다 실행
  • []: 마운트 시에만 실행
  • [dep1, dep2]: 의존성이 변경될 때 실행

클린업 함수

useEffect 내부에서 타이머, 이벤트 리스너, 네트워크 요청 등을 시작했다면 반드시 정리(cleanup)해야 합니다. 그렇지 않으면 메모리 누수가 발생할 수 있습니다.

function Timer() {
  const [seconds, setSeconds] = useState(0);

  useEffect(() => {
    const interval = setInterval(() => {
      setSeconds(s => s + 1);
    }, 1000);

    // 클린업 함수: 언마운트 시 또는 다음 effect 실행 전에 호출
    return () => {
      clearInterval(interval);
    };
  }, []); // 빈 배열이므로 마운트/언마운트 시에만

  return <div>Seconds: {seconds}</div>;
}

클린업이 필요한 경우:

  • setInterval, setTimeout
  • 이벤트 리스너
  • WebSocket 연결
  • subscription
  • 외부 라이브러리 정리

데이터 fetching

API 호출은 useEffect에서 자주 사용되는 패턴입니다. 하지만 컴포넌트가 언마운트되거나 userId가 변경되는 상황을 처리해야 합니다.

function UserProfile({ userId }) {
  const [user, setUser] = useState(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState(null);

  useEffect(() => {
    // cleanup을 위한 flag
    let cancelled = false;

    async function fetchUser() {
      try {
        setLoading(true);
        setError(null);
        const response = await fetch(`/api/users/${userId}`);
        const data = await response.json();

        if (!cancelled) {
          setUser(data);
        }
      } catch (err) {
        if (!cancelled) {
          setError(err.message);
        }
      } finally {
        if (!cancelled) {
          setLoading(false);
        }
      }
    }

    fetchUser();

    // 클린업: 컴포넌트 언마운트나 userId 변경 시
    return () => {
      cancelled = true;
    };
  }, [userId]);

  if (loading) return <div>Loading...</div>;
  if (error) return <div>Error: {error}</div>;
  if (!user) return <div>No user found</div>;

  return (
    <div>
      <h1>{user.name}</h1>
      <p>{user.email}</p>
    </div>
  );
}

이벤트 리스너

function WindowSize() {
  const [size, setSize] = useState({
    width: window.innerWidth,
    height: window.innerHeight
  });

  useEffect(() => {
    const handleResize = () => {
      setSize({
        width: window.innerWidth,
        height: window.innerHeight
      });
    };

    window.addEventListener('resize', handleResize);

    // 클린업
    return () => {
      window.removeEventListener('resize', handleResize);
    };
  }, []);

  return (
    <div>
      Width: {size.width}px, Height: {size.height}px
    </div>
  );
}

localStorage 동기화

function useLocalStorage(key, initialValue) {
  // lazy initialization으로 localStorage 읽기
  const [value, setValue] = useState(() => {
    try {
      const item = window.localStorage.getItem(key);
      return item ? JSON.parse(item) : initialValue;
    } catch (error) {
      console.error(error);
      return initialValue;
    }
  });

  // value가 변경될 때 localStorage에 저장
  useEffect(() => {
    try {
      window.localStorage.setItem(key, JSON.stringify(value));
    } catch (error) {
      console.error(error);
    }
  }, [key, value]);

  return [value, setValue];
}

// 사용
function App() {
  const [name, setName] = useLocalStorage('name', '');
  const [theme, setTheme] = useLocalStorage('theme', 'light');

  return (
    <div>
      <input
        value={name}
        onChange={(e) => setName(e.target.value)}
      />
      <button onClick={() => setTheme(theme === 'light' ? 'dark' : 'light')}>
        Toggle Theme
      </button>
    </div>
  );
}

useRef - 값 유지하기

useRef는 두 가지 주요 용도가 있습니다: (1) DOM 요소에 직접 접근하기, (2) 렌더링과 무관하게 값을 저장하기.

DOM 접근

import { useRef, useEffect } from 'react';

function AutoFocusInput() {
  const inputRef = useRef(null);

  useEffect(() => {
    // 마운트 시 자동 포커스
    inputRef.current.focus();
  }, []);

  return <input ref={inputRef} />;
}

// 실무 예시: 스크롤 제어
function ScrollToTop() {
  const topRef = useRef(null);

  const scrollToTop = () => {
    topRef.current?.scrollIntoView({ behavior: 'smooth' });
  };

  return (
    <div>
      <div ref={topRef}>Top of the page</div>
      {/* 긴 컨텐츠 */}
      <button onClick={scrollToTop}>Scroll to Top</button>
    </div>
  );
}

이전 값 저장

useRef의 강력한 활용법 중 하나는 '이전 값'을 기억하는 것입니다. ref는 리렌더링되어도 값이 유지되기 때문에 가능합니다.

function usePrevious(value) {
  const ref = useRef();

  useEffect(() => {
    ref.current = value;
  }, [value]);

  return ref.current;
}

// 사용
function Counter() {
  const [count, setCount] = useState(0);
  const prevCount = usePrevious(count);

  return (
    <div>
      <p>Current: {count}</p>
      <p>Previous: {prevCount}</p>
      <button onClick={() => setCount(c => c + 1)}>+</button>
    </div>
  );
}

값 변경 감지하지 않기

useState와 달리 useRef는 값이 변경되어도 리렌더링을 일으키지 않습니다. 이 특성을 활용하면 클로저 문제를 해결할 수 있습니다.

function Timer() {
  const [count, setCount] = useState(0);
  const countRef = useRef(count);

  // ref 업데이트는 리렌더링을 발생시키지 않음
  countRef.current = count;

  useEffect(() => {
    const interval = setInterval(() => {
      // 최신 count 값 사용 (클로저 문제 해결)
      console.log('Current count:', countRef.current);
    }, 1000);

    return () => clearInterval(interval);
  }, []); // 의존성 배열이 비어있어도 최신 값 접근 가능

  return (
    <div>
      <p>Count: {count}</p>
      <button onClick={() => setCount(c => c + 1)}>+</button>
    </div>
  );
}

렌더링 횟수 추적

function RenderCounter() {
  const renderCount = useRef(0);

  useEffect(() => {
    renderCount.current += 1;
  });

  return <div>Render count: {renderCount.current}</div>;
}

useContext - 전역 상태 관리

props를 여러 단계를 거쳐 전달하는 것을 'props drilling'이라고 합니다. useContext를 사용하면 이를 해결하고 전역 상태를 쉽게 공유할 수 있습니다.

기본 사용법

import { createContext, useContext, useState } from 'react';

// Context 생성
const ThemeContext = createContext();

// Provider 컴포넌트
function ThemeProvider({ children }) {
  const [theme, setTheme] = useState('light');

  const toggleTheme = () => {
    setTheme(prev => prev === 'light' ? 'dark' : 'light');
  };

  return (
    <ThemeContext.Provider value={{ theme, toggleTheme }}>
      {children}
    </ThemeContext.Provider>
  );
}

// Hook으로 Context 사용
function useTheme() {
  const context = useContext(ThemeContext);
  if (!context) {
    throw new Error('useTheme must be used within ThemeProvider');
  }
  return context;
}

// 컴포넌트에서 사용
function ThemedButton() {
  const { theme, toggleTheme } = useTheme();

  return (
    <button
      onClick={toggleTheme}
      style={{
        background: theme === 'light' ? '#fff' : '#333',
        color: theme === 'light' ? '#000' : '#fff'
      }}
    >
      Toggle Theme (Current: {theme})
    </button>
  );
}

// App
function App() {
  return (
    <ThemeProvider>
      <ThemedButton />
    </ThemeProvider>
  );
}

인증 Context

실무에서 가장 많이 사용되는 패턴 중 하나는 인증(Authentication) Context입니다. 로그인 상태를 전역적으로 관리할 수 있습니다.

const AuthContext = createContext();

function AuthProvider({ children }) {
  const [user, setUser] = useState(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    // 초기 인증 상태 확인
    checkAuth();
  }, []);

  async function checkAuth() {
    try {
      const response = await fetch('/api/auth/me');
      if (response.ok) {
        const user = await response.json();
        setUser(user);
      }
    } catch (error) {
      console.error('Auth check failed:', error);
    } finally {
      setLoading(false);
    }
  }

  async function login(email, password) {
    const response = await fetch('/api/auth/login', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ email, password })
    });

    if (response.ok) {
      const user = await response.json();
      setUser(user);
      return user;
    }

    throw new Error('Login failed');
  }

  async function logout() {
    await fetch('/api/auth/logout', { method: 'POST' });
    setUser(null);
  }

  const value = {
    user,
    loading,
    login,
    logout,
    isAuthenticated: !!user
  };

  return (
    <AuthContext.Provider value={value}>
      {children}
    </AuthContext.Provider>
  );
}

function useAuth() {
  const context = useContext(AuthContext);
  if (!context) {
    throw new Error('useAuth must be used within AuthProvider');
  }
  return context;
}

// 보호된 라우트
function ProtectedRoute({ children }) {
  const { isAuthenticated, loading } = useAuth();

  if (loading) return <div>Loading...</div>;
  if (!isAuthenticated) return <Navigate to="/login" />;

  return children;
}
